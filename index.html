import { signInAnonymously, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-auth.js"; // دخول تلقائي بدون بريد أو كلمة مرور let user = null; onAuthStateChanged(auth, async (currentUser) => { try { if (currentUser) { user = currentUser; document.getElementById("login").classList.add("hidden"); document.getElementById("app").classList.remove("hidden"); await refreshData(); } else { await signInAnonymously(auth); } } catch (error) { console.error("Firebase error:", error); alert("تعذر الاتصال بفايربيس: " + error.message); } });
